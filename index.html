SELECT p.nome,
       c.nome_cargo,
       f.salario_vigente
FROM tb_funcionarios f
JOIN tb_pessoas p
  ON f.pk_pessoa_id = p.pk_pessoa_id
JOIN tb_cargos c
  ON f.fk_cargo = c.pk_cargo;
